83inline bool append_osc_argument(
84 oscpack::OutboundPacketStream& p, char tag, const QJSValue& v)
85{
86 switch(tag)
87 {
88 case 'i':
89 // ToInt32 semantics, as everywhere else a script hands QJSValue an int.
90 p << (int32_t)v.toInt();
91 return true;
92 case 'h':
93 p << (int64_t)v.toNumber();
94 return true;
95 case 'f':
96 p << (float)v.toNumber();
97 return true;
98 case 'd':
99 p << (double)v.toNumber();
100 return true;
101 case 's': {
102 const auto str = v.toString().toUtf8();
103 p << oscpack::string_view{str.constData(), (std::size_t)str.size()};
104 return true;
105 }
106 case 'S': {
107 const auto str = v.toString().toUtf8();
108 p << oscpack::Symbol{str.constData()};
109 return true;
110 }
111 case 'c': {
112 const auto str = v.toString();
113 p << (char)(str.isEmpty() ? '\0' : str[0].toLatin1());
114 return true;
115 }
116 case 'b': {
117 const auto blob = v.toVariant().toByteArray();
118 p << oscpack::Blob{
119 blob.constData(), (oscpack::osc_bundle_element_size_t)blob.size()};
120 return true;
121 }
122 case 'T':
123 p << true;
124 return true;
125 case 'F':
126 p << false;
127 return true;
128 case 'I':
129 p << oscpack::Infinitum();
130 return true;
131 default:
132 return false;
133 }
134}

156template <typename Writer>
157void write_osc_message(
158 Writer&& write, const QByteArray& address, const QJSValueList& values,
159 const QByteArray& typetags)
160{
161 const std::string addr = address.toStdString();
162 // No parameter and no unit here: this is a raw socket, not a device tree
163 // address, so nothing dataspace-dependent can apply.
164 const ossia::unit_t no_unit{};
165
166 auto& pool = ossia::buffer_pool::instance();
167 auto buf = pool.acquire();
168
169 while(buf.size() < ossia::net::max_osc_message_size)
170 {
171 try
172 {
173 oscpack::OutboundPacketStream str{buf.data(), buf.size()};
174 str << oscpack::BeginMessageN(addr);
175
176 const ossia::net::osc_1_0_outbound_dynamic_policy untyped{{str, no_unit}};
177 for(int i = 0; i < values.size(); i++)
178 {
179 const char tag = i < typetags.size() ? typetags[i] : '\0';
180 if(!append_osc_argument(str, tag, values[i]))
181 ossia::qt::value_from_js(values[i]).apply(untyped);
182 }
183 str << oscpack::EndMessage();
184
185 write(str.Data(), str.Size());
186 break;
187 }
188 catch(...)
189 {
190 // oscpack throws when the buffer is too small: grow and retry, as the
191 // untyped list path does, and give up at max_osc_message_size.
192 const auto n = buf.size();
193 buf.clear();
194 buf.resize(n * 2 + 1);
195 }
196 }
197
198 pool.release(std::move(buf));
199}

220 {
221 auto& sock = *self.socket;
222 using socket_type = std::remove_cvref_t<decltype(sock)>;
223 using writer_type = ossia::net::socket_writer<socket_type>;
224 using send_visitor = ossia::net::osc_value_send_visitor<
225 ossia::net::full_parameter_data, ossia::net::osc_1_0_policy, writer_type>;
226
227 ossia::net::full_parameter_data p;
228 const std::string addr = address.toStdString();
229
230 switch(values.size())
231 {
232 case 0: {
233 ossia::value{ossia::impulse{}}.apply(
234 send_visitor{p, addr, writer_type{sock}});
235 break;
236 }
237 case 1: {
238 auto v = ossia::qt::value_from_js(values[0]);
239 v.apply(send_visitor{p, addr, writer_type{sock}});
240 break;
241 }
242 default: {

244 vec.reserve(values.size());
245 for(const auto& v : values)
246 vec.push_back(ossia::qt::value_from_js(v));
247 ossia::value vvec(std::move(vec));

287 static QJSValue
288 oscArgToQJSValue(const oscpack::ReceivedMessageArgument& arg, QJSEngine& engine)
289 {
290 switch(arg.TypeTag())
291 {
292 case oscpack::INT32_TYPE_TAG:
293 return QJSValue(arg.AsInt32Unchecked());
294 case oscpack::INT64_TYPE_TAG:
295 return QJSValue((int)arg.AsInt64Unchecked());
296 case oscpack::FLOAT_TYPE_TAG:
297 return QJSValue(arg.AsFloatUnchecked());
298 case oscpack::DOUBLE_TYPE_TAG:
299 return QJSValue(arg.AsDoubleUnchecked());
300 case oscpack::TIME_TAG_TYPE_TAG:
301 return QJSValue((int)arg.AsTimeTagUnchecked());
302 case oscpack::CHAR_TYPE_TAG:
303 return QJSValue(QString(QChar(arg.AsCharUnchecked())));
304 case oscpack::TRUE_TYPE_TAG:
305 return QJSValue(true);
306 case oscpack::FALSE_TYPE_TAG:
307 return QJSValue(false);
308 case oscpack::STRING_TYPE_TAG:
309 return QJSValue(QString::fromUtf8(arg.AsStringUnchecked()));
310 case oscpack::SYMBOL_TYPE_TAG:
311 return QJSValue(QString::fromUtf8(arg.AsSymbolUnchecked()));
312 case oscpack::BLOB_TYPE_TAG: {
313 const void* data{};
314 oscpack::osc_bundle_element_size_t size{};
315 arg.AsBlobUnchecked(data, size);
316 QByteArray blob((const char*)data, size);
317 return engine.toScriptValue(blob);
318 }
319 case oscpack::RGBA_COLOR_TYPE_TAG: {
320 auto c = arg.AsRgbaColorUnchecked();
321 auto array = engine.newArray(4);
322 array.setProperty(0, uint8_t(c >> 24 & 0xFF));
323 array.setProperty(1, uint8_t(c >> 16 & 0xFF));
324 array.setProperty(2, uint8_t(c >> 8 & 0xFF));
325 array.setProperty(3, uint8_t(c & 0xFF));
326 return array;
327 }
328 default:
329 return QJSValue(QJSValue::NullValue);
330 }
