24inline QByteArray bytes_from_js(const QJSValue& v)
25{
26 if(v.isArray())
27 {
28 QByteArray res;
29 const int n = v.property("length").toInt();
30 res.reserve(n);
31 for(int i = 0; i < n; i++)
32 res.append(char(v.property(i).toInt()));
33 return res;
34 }
35 if(v.isString())
36 return v.toString().toUtf8();
37 return v.toVariant().toByteArray();

76 {
77 for(;;)

79 pending p;

82 queue_cv.wait(lock, [this] { return stopping || !queue.empty(); });
83 if(stopping)

88 p = std::move(queue.front());

94 std::size_t(p.data.size())};
95 const int r = p.feature ? device.send_feature_report(bytes) : device.write(bytes);
96 if(r < 0)

100 if(auto s = w.lock(); s && s->self)

104 }
