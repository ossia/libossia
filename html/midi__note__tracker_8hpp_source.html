74 void track(const libremidi::ump* it, const libremidi::ump* end) noexcept
75 {
76 uint32_t cur_groups = groups;
77 uint32_t cur_damper = damper;
78 bool cur_midi1 = midi1;
79
80 for(; it != end; ++it)
81 {
82 const uint32_t w = it->data[0];
83 const uint32_t type = w >> 28;
84 // MIDI 1 and MIDI 2 channel voice messages share the layout of their
85 // first word up to the pitch / controller index.
86 constexpr uint32_t channel_voice = (1u << CMIDI2_MESSAGE_TYPE_MIDI_1_CHANNEL)
87 | (1u << CMIDI2_MESSAGE_TYPE_MIDI_2_CHANNEL);
88 if(!((channel_voice >> type) & 1))
89 continue;
90
91 const bool is_midi1 = type == CMIDI2_MESSAGE_TYPE_MIDI_1_CHANNEL;
92 const uint32_t status = (w >> 20) & 0xF;
93 if((status & 0xE) == 0x8)
94 {
95 // In MIDI 1 a note on of velocity zero is a note off; in MIDI 2 it
96 // is a note.
97 const bool on = (status & 1) && !(is_midi1 && (w & 0x7F) == 0);
98
99 // channel * 128 + pitch
100 const uint32_t key = ((w >> 9) & 0x780) | ((w >> 8) & 0x7F);
101 uint64_t& word = notes[key >> 6];
102 const uint64_t bit = uint64_t(1) << (key & 63);
103 word = on ? (word | bit) : (word & ~bit);
104 if(on)
105 {
106 cur_groups |= 1u << ((w >> 24) & 0xF);
107 cur_midi1 = is_midi1;
108 }
109 }
110 else if(status == 0xB && ((w >> 8) & 0x7F) == 64)
111 {
112 const uint32_t channel = (w >> 16) & 0xF;
113 const bool down = is_midi1 ? (w & 0x7F) >= 64 : it->data[1] >= 0x80000000u;
114 cur_damper = (cur_damper & ~(1u << channel)) | (uint32_t(down) << channel);
115 if(down)
116 {
117 cur_groups |= 1u << ((w >> 24) & 0xF);
118 cur_midi1 = is_midi1;
119 }
120 }
121 }
122
123 groups = uint16_t(cur_groups);
124 damper = uint16_t(cur_damper);
125 midi1 = cur_midi1;
126 }

134 if(empty())

136
137 for(uint32_t gs = groups; gs != 0; gs &= gs - 1)
138 {
139 const auto group = uint8_t(std::countr_zero(gs));
140

142 {
143 const auto channel = uint8_t(std::countr_zero(cs));
144 push(
145 out, timestamp, midi1,

147 : cmidi2_ump_midi2_cc(group, channel, 64, 0));
148 }
149
150 for(uint8_t channel = 0; channel < channels; channel++)
151 {
152 for(int half = 0; half < 2; half++)
153 {
154 for(uint64_t bits = notes[channel * 2 + half]; bits != 0; bits &= bits - 1)
155 {
156 const auto pitch = uint8_t(half * 64 + std::countr_zero(bits));
157 push(
158 out, timestamp, midi1,
159 midi1 ? int64_t(cmidi2_ump_midi1_note_off(group, channel, pitch, 0))
160 : cmidi2_ump_midi2_note_off(group, channel, pitch, 0, 0, 0));
161 }
162 }
